\section{Introduction}
\label{sec:introduction}
%======================================================================

The discrete dipole approximation (DDA) is a flexible numerical method for
computing electromagnetic scattering and absorption by particles of arbitrary
shape and composition~\cite{Purcell1973,Draine1988}.
The target particle is represented by an array of $N$ point dipoles on a cubic
lattice with spacing~$d$.  Each dipole acquires a polarisation in response to
the local electric field, which consists of the incident wave plus the fields
radiated by all other dipoles.

This document describes the theoretical background and the algorithms
that are implemented in version~0.8.1 of
\texttt{block-DDA\_Py}, covering:
\begin{enumerate}
  \item the DDA formulation and polarisability prescription
        (Sec.~\ref{sec:dda-formulation}),
  \item the Goodman~(1991) FFT-accelerated matrix--vector product
        (Sec.~\ref{sec:goodman-fft}),
  \item the block-Krylov iterative solvers
        (Sec.~\ref{sec:block-krylov}),
  \item the shifted block GMRES for refractive-index sweeps
        (Sec.~\ref{sec:shifted-gmres}),
  \item the scattering observables computed from the solution
        (Sec.~\ref{sec:observables}),
  \item the Gaussian Random Ellipsoid (GRE) shape model
        (Sec.~\ref{sec:gre}),
  \item the spheroid parameter sweep and its HDF5 output format
        (Sec.~\ref{sec:spheroid-sweep}),
  \item the Mie theory reference solution for homogeneous spheres
        (Sec.~\ref{sec:mie}).
\end{enumerate}

Throughout this note, the electromagnetic equations are written in
Gaussian (CGS) units; lengths are expressed in micrometers (\si{\micro\metre}).
Vectors are set in bold italic ($\bm{E}$, $\bm{k}$, $\bm{r}$),
matrices and second-rank tensors in upright bold ($\mathbf{A}$,
$\boldsymbol{\alpha}$), and unit vectors carry a hat
($\hat{\bm{e}}$, $\hat{\bm{r}}$).

\paragraph{Companion code for high-contrast particles.}
DDA is most efficient when the refractive-index contrast is moderate
($|m_p/m_m| \lesssim 2$). For high-contrast targets --- e.g.\ plasmonic
metals (gold, silver), iron-oxide aggregates, extreme aspect ratios, or
surface roughness finer than the DDA lattice spacing --- the DDA
iteration converges slowly (or not at all) and the discretization error
is dominated by the staircasing of curved boundaries on the cubic
lattice. In those regimes the companion code
\texttt{block-VIEM.jl}~\cite{blockVIEM}, a Julia implementation of the
volume integral equation method (VIEM) with a tetrahedral-mesh
SWG/half-SWG basis, Duffy-transform singular integration, and AIM-FFT
accelerated block-Krylov iteration, is recommended. The two codes
share the same physical input conventions ($\lambda_0$, $m_m$, $m_p$,
including anisotropic $m_{p,xyz}$), Euler-angle convention (ZYZ
intrinsic), GRE shape parametrization, HDF5 output schema, and CAS-v2
observable definitions, so downstream analysis pipelines can consume
either output without modification. Practically, we recommend
\texttt{block-DDA\_Py} for $|m_p/m_m| < 2$ (where DDA is
100--1000$\times$ faster in setup cost) and \texttt{block-VIEM.jl}
otherwise.

\section{DDA Formulation}
\label{sec:dda-formulation}
%======================================================================

%----------------------------------------------------------------------
\subsection{Fundamental equation}
\label{sec:fundamental-eq}
%----------------------------------------------------------------------

Consider a monochromatic plane wave of vacuum wavelength $\lambda_0$ incident
on a particle embedded in a non-absorbing medium of refractive index~$m_m$.
The wavenumber in the medium is
\begin{equation}
  k = \frac{2\pi\, m_m}{\lambda_0}.
  \label{eq:wavenumber}
\end{equation}

The particle is discretised into $N$ cubic elements (dipoles) of side
length~$d$ and volume $v = d^3$, placed on a regular lattice with grid
dimensions $N_x \times N_y \times N_z$.
The lattice position of the $i$-th dipole is $\bm{r}_i$.

The polarisation $\bm{P}_i$ of dipole~$i$ satisfies
\begin{equation}
  \bm{P}_i = \boldsymbol{\alpha}_i \cdot \bm{E}(\bm{r}_i),
  \label{eq:polarisation}
\end{equation}
where $\boldsymbol{\alpha}_i$ is the $3\times 3$ polarisability tensor and
$\bm{E}(\bm{r}_i)$ is the total electric field at position~$\bm{r}_i$.
The total field is the superposition of the incident field and the
contributions from all other dipoles:
\begin{equation}
  \bm{E}(\bm{r}_i)
    = \bm{E}_{\text{inc},i}
      - \sum_{j \neq i} \mathbf{G}(\bm{r}_i - \bm{r}_j) \cdot \bm{P}_j,
  \label{eq:total-field}
\end{equation}
where $\mathbf{G}$ is the free-space dyadic Green's function
(Sec.~\ref{sec:green-tensor}).
Substituting \eqref{eq:total-field} into \eqref{eq:polarisation} and
rearranging yields the $3N \times 3N$ linear system
\begin{equation}
  \mathbf{A} \, \bm{P} = \bm{E}_{\text{inc}},
  \label{eq:dda-system}
\end{equation}
where $\bm{P}$ and $\bm{E}_{\text{inc}}$ are column vectors of length~$3N$
formed by concatenating the three Cartesian components of each dipole
(ordering: $P_{x,0}, P_{y,0}, P_{z,0}, P_{x,1}, \ldots$),
and $\mathbf{A}$ is a complex-valued matrix with
\begin{equation}
  \mathbf{A}_{ij} =
  \begin{cases}
    \boldsymbol{\alpha}_i^{-1}         & \text{if } i = j, \\[4pt]
    \mathbf{G}(\bm{r}_i - \bm{r}_j)   & \text{if } i \neq j.
  \end{cases}
  \label{eq:A-matrix}
\end{equation}

When the dipoles are placed on a regular cubic lattice, the off-diagonal
blocks $\mathbf{A}_{ij}$ depend only on the difference $\bm{r}_i - \bm{r}_j$,
giving $\mathbf{A}$ a block-Toeplitz structure that can be exploited by FFT
(Sec.~\ref{sec:goodman-fft}).

%----------------------------------------------------------------------
\subsection{Incident field}
\label{sec:incident-field}
%----------------------------------------------------------------------

The incident plane wave in the medium is
\begin{equation}
  \bm{E}_{\text{inc},i}
    = \bm{E}_0 \exp(i \bm{k} \cdot \bm{r}_i - i\omega t),
  \label{eq:incident-wave}
\end{equation}
where $\bm{k} = k\,\hat{\bm{u}}_{\text{inc}}$ with
$\hat{\bm{u}}_{\text{inc}}$ the propagation direction
(fixed to $+z$ in the laboratory frame).
The time-harmonic factor $e^{-i\omega t}$ is suppressed hereafter.

To simulate orientation-averaged scattering, the particle is rotated through
$L = N_\alpha \times N_\beta \times N_\gamma$ orientations on a
deterministic grid of intrinsic ZYZ Euler angles:
\begin{itemize}
  \item $\alpha$: $N_\alpha$ equally spaced values in $[0, 2\pi)$,
  \item $\beta$: $N_\beta$ values chosen so that $\cos\beta$ is equally
        spaced in $(-1, 1)$ (equal-area midpoints on the sphere),
  \item $\gamma$: $N_\gamma$ equally spaced values in $[0, 2\pi)$.
\end{itemize}
This deterministic scheme avoids sampling noise and ensures reproducibility.
For each orientation, the laboratory-frame vectors are transformed into
the particle frame via the inverse rotation matrix.
Batch rotation is performed using \texttt{scipy.spatial.transform.Rotation}.

Following the CAS-v2 (Complex Amplitude Sensing, version~2) beam
configuration, the incident polarisation is assumed to be left-handed
circular.
This setting is hard-coded in the current version (v0.6.0).
The polarisation vector is:
\begin{equation}
  \hat{\bm{e}}_0
    = \frac{1}{\sqrt{2}}\,\hat{\bm{e}}_\theta
      + \frac{i}{\sqrt{2}}\,\hat{\bm{e}}_\phi,
  \label{eq:circular-pol}
\end{equation}
where $\hat{\bm{e}}_\theta$ and $\hat{\bm{e}}_\phi$ are the polar and
azimuthal unit vectors associated with the incident direction.

%----------------------------------------------------------------------
\subsection{Free-space dyadic Green's function}
\label{sec:green-tensor}
%----------------------------------------------------------------------

The $3\times 3$ interaction tensor for two dipoles separated by
$\bm{r} = \bm{r}_i - \bm{r}_j$ ($r = |\bm{r}|$, $\hat{\bm{r}} = \bm{r}/r$)
is~\cite{Goodman1991}
\begin{equation}
  \mathbf{G}(\bm{r})
    = \frac{e^{ikr}}{r}
      \left[
        k^2 \bigl(\hat{\bm{r}} \otimes \hat{\bm{r}} - \mathbf{I}_3\bigr)
        + \frac{1 - ikr}{r^2}
          \bigl(\mathbf{I}_3 - 3\,\hat{\bm{r}} \otimes \hat{\bm{r}}\bigr)
      \right],
  \label{eq:green-tensor}
\end{equation}
where $\otimes$ denotes the outer (tensor) product and $\mathbf{I}_3$ the
$3\times 3$ identity matrix.
The self-interaction ($\bm{r} = \bm{0}$) is excluded from
$\mathbf{G}$ and handled separately via the diagonal
$\boldsymbol{\alpha}_i^{-1}$ in \eqref{eq:A-matrix}.

%----------------------------------------------------------------------
\subsection{Polarisability: Clausius--Mossotti with radiative correction}
\label{sec:polarisability}
%----------------------------------------------------------------------

The relative permittivity of the $i$-th dipole along Cartesian axis~$c$
($c \in \{x, y, z\}$) is
\begin{equation}
  \epsilon_{r,c} = \frac{m_{p,c}^2}{m_m^2},
  \label{eq:eper-r}
\end{equation}
where $m_{p,c}$ is the (generally complex) particle refractive index along
axis~$c$.

The static Clausius--Mossotti (CM) polarisability is
\begin{equation}
  \alpha_{0,c}
    = \frac{3}{4\pi}\,
      \frac{\epsilon_{r,c} - 1}{\epsilon_{r,c} + 2}\; v,
  \label{eq:cm-polarisability}
\end{equation}
where $v = d^3$ is the dipole volume.

To improve accuracy, the CM polarisability is corrected for radiative
reaction following Chaumet \& Rahmani~\cite{Chaumet2009}.
Define the equivalent-volume sphere radius
\begin{equation}
  a = \left(\frac{3v}{4\pi}\right)^{\!1/3}
  \label{eq:equiv-radius-a}
\end{equation}
and the correction term
\begin{equation}
  M = \frac{8\pi}{3}
      \bigl[(1 - ika)\,e^{ika} - 1\bigr].
  \label{eq:M-term}
\end{equation}
The dynamical (radiative-reaction corrected) polarisability is then
\begin{equation}
  \alpha_c
    = \frac{\alpha_{0,c}}{1 - M\,\alpha_{0,c}\,/\,v}.
  \label{eq:rr-polarisability}
\end{equation}

In general the particle may be optically anisotropic, so
$\boldsymbol{\alpha}_i = \operatorname{diag}(\alpha_x, \alpha_y, \alpha_z)$.
For isotropic particles, $m_{p,x} = m_{p,y} = m_{p,z}$ and the
polarisability reduces to a scalar times $\mathbf{I}_3$.

%======================================================================
\section{FFT-Accelerated Matrix--Vector Product (Goodman 1991)}
\label{sec:goodman-fft}
%======================================================================

Direct evaluation of the matrix--vector product (MVP)
$\mathbf{A}\,\bm{P}$ requires $O(N^2)$ operations and $O(N^2)$ storage.
Because the off-diagonal part of $\mathbf{A}$ has block-Toeplitz structure
on a regular cubic lattice, the MVP can be evaluated via 3D FFT in
$O(N \log N)$ time with $O(N)$ storage~\cite{Goodman1991}.

%----------------------------------------------------------------------
\subsection{Interaction tensor on the doubled grid}
\label{sec:doubled-grid}
%----------------------------------------------------------------------

The physical cuboid lattice has dimensions $N_x \times N_y \times N_z$.
The index difference along each axis~$\alpha$ ranges from
$-(N_\alpha - 1)$ to $+(N_\alpha - 1)$.
To evaluate the linear convolution as a cyclic convolution via FFT,
the lattice is extended to $2N_x \times 2N_y \times 2N_z$ with the
following index mapping.

For each extended-grid index $I_\alpha \in \{0, 1, \ldots, 2N_\alpha - 1\}$,
the physical integer displacement $\Delta\alpha$ is
\begin{equation}
  \Delta\alpha(I_\alpha) =
  \begin{cases}
    I_\alpha            & 0 \le I_\alpha \le N_\alpha - 1
                          \quad\text{(positive displacements)}, \\
    0                   & I_\alpha = N_\alpha
                          \quad\text{(Nyquist; zeroed out)}, \\
    I_\alpha - 2N_\alpha & N_\alpha + 1 \le I_\alpha \le 2N_\alpha - 1
                          \quad\text{(negative displacements)}.
  \end{cases}
  \label{eq:index-mapping}
\end{equation}

The physical displacement vector at extended-grid index
$(I_x, I_y, I_z)$ is
$\bm{r} = d\,(\Delta x,\, \Delta y,\, \Delta z)$,
and the interaction tensor $\mathbf{G}(\bm{r})$ is computed via
\eqref{eq:green-tensor}.
Three special cases are set to zero:
\begin{itemize}
  \item self-interaction at the origin:
        $\mathbf{G}(\bm{0}) = \mathbf{0}$,
  \item Nyquist positions ($I_\alpha = N_\alpha$ for any~$\alpha$):
        physically unreachable displacements.
\end{itemize}

The resulting array has shape
$(2N_x, 2N_y, 2N_z, 3, 3)$ and dtype \texttt{complex128}.

%----------------------------------------------------------------------
\subsection{Pre-computation of the FFT}
\label{sec:fft-precompute}
%----------------------------------------------------------------------

Before the iterative solver begins, the 3D discrete Fourier transform of
$\mathbf{G}$ along the spatial axes is computed once:
\begin{equation}
  \widetilde{\mathbf{G}}
    = \mathrm{FFT}_{3\text{D}}\!\bigl[\mathbf{G}\bigr].
  \label{eq:fft-precompute}
\end{equation}
$\widetilde{\mathbf{G}}$ has the same shape
$(2N_x, 2N_y, 2N_z, 3, 3)$ and is stored in memory throughout the
simulation.
It is independent of the incident direction, polarisation, and Krylov
iteration, so this computation is performed only once per
$(\lambda_0, m_m)$ pair.

%----------------------------------------------------------------------
\subsection{Block MVP via FFT convolution}
\label{sec:fft-mvp}
%----------------------------------------------------------------------

Given a block of $L$ polarisation vectors stacked as columns in
$\mathbf{P}_{\text{block}}$ (shape $3N_xN_yN_z \times L$), the
off-diagonal MVP $\mathbf{A}_{\text{off}} \, \mathbf{P}_{\text{block}}$
is computed as follows.

\begin{enumerate}
  \item \textbf{Reshape} $\mathbf{P}_{\text{block}}$ from
        $(3 N_x N_y N_z,\, L)$ to the 5D physical representation
        $(N_x, N_y, N_z, 3, L)$.
  \item \textbf{Zero-pad} to the doubled grid:
        \begin{equation}
          \hat{\mathbf{P}}(I_x, I_y, I_z, c, \ell) =
          \begin{cases}
            \mathbf{P}(I_x, I_y, I_z, c, \ell)
              & \text{if } I_\alpha < N_\alpha \;\forall \alpha, \\
            0 & \text{otherwise}.
          \end{cases}
          \label{eq:zero-pad}
        \end{equation}
  \item \textbf{Forward 3D FFT} along spatial axes:
        $\widetilde{\mathbf{P}} = \mathrm{FFT}_{3\text{D}}[\hat{\mathbf{P}}]$.
  \item \textbf{Pointwise tensor product} in Fourier space:
        \begin{equation}
          \widetilde{Y}_{i,\ell}
            = \sum_{j=1}^{3}
              \widetilde{G}_{ij}\;\widetilde{P}_{j,\ell}
          \qquad
          \text{(at each spatial frequency)},
          \label{eq:fourier-product}
        \end{equation}
        i.e.\ a $3\times 3$ matrix times a $3\times L$ matrix at every
        grid point.
  \item \textbf{Inverse 3D FFT}:
        $\hat{\mathbf{Y}} = \mathrm{FFT}_{3\text{D}}^{-1}[\widetilde{\mathbf{Y}}]$.
  \item \textbf{Extract} the physical region
        $\hat{\mathbf{Y}}[{:}N_x,\, {:}N_y,\, {:}N_z,\, :,\, :]$
        and flatten back to $(3 N_x N_y N_z, L)$.
\end{enumerate}

The full preconditioned MVP including the diagonal is
\begin{equation}
  \mathbf{A}\,\bm{S}
    = \operatorname{diag}(\mathbf{A})\cdot\bm{S}
      + \mathbf{A}_{\text{off}}\,\bm{S},
  \label{eq:full-mvp}
\end{equation}
where $\operatorname{diag}(\mathbf{A}) = \boldsymbol{\alpha}^{-1}$ is applied
element-wise and $\mathbf{A}_{\text{off}}$ is evaluated via the FFT
procedure above.

%----------------------------------------------------------------------
\subsection{Computational complexity and memory}
\label{sec:complexity}
%----------------------------------------------------------------------

\begin{itemize}
  \item \textbf{Time}: Each MVP costs $O(N_{\text{cuboid}} \log N_{\text{cuboid}})$
        where $N_{\text{cuboid}} = N_x N_y N_z$.
        For the block variant with $L$ right-hand sides, the $L$ columns
        are processed in a single batched FFT call.
  \item \textbf{Memory}: The dominant allocations are:
  \begin{itemize}
    \item $\widetilde{\mathbf{G}}$:
          $2N_x \times 2N_y \times 2N_z \times 9$ complex128 entries
          $= 1152\,N_{\text{cuboid}}$ bytes (static, allocated once);
    \item padded polarisation arrays (forward and inverse FFT):
          $2N_x \times 2N_y \times 2N_z \times 3 \times L$ complex128
          entries each $= 384\,L\,N_{\text{cuboid}}$ bytes (two such arrays
          at peak).
  \end{itemize}
  Peak memory $\approx (1152 + 768\,L)\,N_{\text{cuboid}}$ bytes.
\end{itemize}

%----------------------------------------------------------------------
\subsection{Multi-threaded FFT}
\label{sec:fft-workers}
%----------------------------------------------------------------------

The 3D FFTs are computed by \texttt{scipy.fft.fftn} with multi-threading.
The default number of worker threads is
$\max(1,\; n_{\text{cpu}} - 2)$, leaving two cores free for other
processes.
This can be overridden at runtime via the environment variable
\texttt{DDA\_FFT\_WORKERS}.

%======================================================================
\section{Block-Krylov Iterative Solvers}
\label{sec:block-krylov}
%======================================================================

The DDA system \eqref{eq:dda-system} is solved for $L$ right-hand sides
simultaneously.
Stacking the $L$ incident-field vectors as columns of a block matrix
$\mathbf{B}$ (shape $3N \times L$), the problem becomes
\begin{equation}
  \mathbf{A}\,\mathbf{X} = \mathbf{B},
  \label{eq:block-system}
\end{equation}
where $\mathbf{X}$ (shape $3N \times L$) contains the $L$ solution
vectors.

The solver of \texttt{block-DDA\_Py} is unrestarted block GMRES
without preconditioning (Sec.~\ref{sec:block-gmres}).
The method \texttt{DiscreteDipoles.solve\_matrix\_equation} calls it.
Earlier versions used Block-BiCGSTAB with Jacobi preconditioning.
That solver and two other block solvers are still available in
\texttt{bl\_krylov/bl\_krylov.py} (Sec.~\ref{sec:alt-solvers}).

%----------------------------------------------------------------------
\subsection{Block GMRES}
\label{sec:block-gmres}
%----------------------------------------------------------------------

Block GMRES~\cite{SaadSchultz1986,SimonciniGallopoulos1996} builds an
orthonormal basis of the block Krylov subspace
\begin{equation}
  \mathcal{K}_n(\mathbf{A}, \mathbf{B})
    = \operatorname{span}\{\mathbf{B}, \mathbf{A}\mathbf{B}, \ldots,
      \mathbf{A}^{n-1}\mathbf{B}\} .
  \label{eq:block-krylov-subspace}
\end{equation}
Its iterate is the element of this subspace that minimises the
Frobenius norm of the residual.
The implementation is \texttt{bl\_gmres\_mvp\_fft}.
It is the multi-orientation block GMRES of Ref.~\cite{Moteki2026}.

\paragraph{Block Arnoldi process.}
The initial guess is $\mathbf{X}^{(0)} = \mathbf{0}$, so the initial
residual is $\mathbf{B}$.
Its thin QR factorisation is
$\mathbf{B} = \mathbf{V}_1 \boldsymbol{\Lambda}$ with
$\mathbf{V}_1 \in \mathbb{C}^{3N \times L}$ and
$\boldsymbol{\Lambda} \in \mathbb{C}^{L \times L}$.
Step $k$ computes one block matrix--vector product
$\mathbf{W} = \mathbf{A}\,\mathbf{V}_k$ (Sec.~\ref{sec:goodman-fft}).
The block $\mathbf{W}$ is then orthogonalised against the previous
blocks by the block modified Gram--Schmidt process:
\begin{equation}
  \mathbf{H}_{ik} = \mathbf{V}_i^\dagger\,\mathbf{W},
  \qquad
  \mathbf{W} \leftarrow \mathbf{W} - \mathbf{V}_i\,\mathbf{H}_{ik},
  \qquad i = 1, \ldots, k .
  \label{eq:block-gram-schmidt}
\end{equation}
The thin QR factorisation
$\mathbf{W} = \mathbf{V}_{k+1}\,\mathbf{H}_{k+1,k}$ gives the next
block.
After $n$ steps the blocks satisfy
\begin{equation}
  \mathbf{A}\,\mathbf{Q}_n = \mathbf{Q}_{n+1}\,\bar{\mathbf{H}}_n,
  \qquad
  \mathbf{Q}_n = [\mathbf{V}_1, \ldots, \mathbf{V}_n],
  \label{eq:block-arnoldi-full}
\end{equation}
where $\bar{\mathbf{H}}_n \in \mathbb{C}^{(n+1)L \times nL}$ is the
block upper Hessenberg matrix of the blocks $\mathbf{H}_{ik}$.

\paragraph{Iterate and residual.}
The iterate after $n$ steps is
$\mathbf{X}^{(n)} = \mathbf{Q}_n \mathbf{Y}^{(n)}$.
Because $\mathbf{Q}_{n+1}$ has orthonormal columns and
$\mathbf{B} = \mathbf{Q}_{n+1} \mathbf{E}_1 \boldsymbol{\Lambda}$ with
$\mathbf{E}_1 = [\mathbf{I}_L, \mathbf{0}, \ldots, \mathbf{0}]^\top$,
the residual norm is
$\| \mathbf{B} - \mathbf{A}\,\mathbf{Q}_n \mathbf{Y} \|_F
 = \| \mathbf{E}_1 \boldsymbol{\Lambda}
      - \bar{\mathbf{H}}_n \mathbf{Y} \|_F$.
The coefficient block is therefore
\begin{equation}
  \mathbf{Y}^{(n)}
    = \arg\min_{\mathbf{Y}}
      \bigl\| \mathbf{E}_1 \boldsymbol{\Lambda}
        - \bar{\mathbf{H}}_n \mathbf{Y} \bigr\|_F .
  \label{eq:gmres-lsq}
\end{equation}

The least-squares problem \eqref{eq:gmres-lsq} is not solved at every
step.
The matrix $\bar{\mathbf{H}}_n$ is reduced to upper triangular form
one block column at a time.
Step $k$ first applies the orthogonal transformations of the steps
$1, \ldots, k-1$ to the new block column.
It then computes a QR factorisation of the $2L \times L$ block that
holds the diagonal and subdiagonal blocks of this column.
The $2L \times 2L$ orthogonal factor is stored and is also applied to
the transformed right-hand side.
After $n$ steps the right-hand side
$\mathbf{E}_1 \boldsymbol{\Lambda}$ has become
$[\mathbf{g}_n^\top, \boldsymbol{\rho}_n^\top]^\top$ with
$\boldsymbol{\rho}_n \in \mathbb{C}^{L \times L}$, and
\begin{equation}
  \frac{\| \mathbf{B} - \mathbf{A}\,\mathbf{X}^{(n)} \|_F}
       {\| \mathbf{B} \|_F}
  = \frac{\| \boldsymbol{\rho}_n \|_F}{\| \mathbf{B} \|_F} .
  \label{eq:gmres-residual}
\end{equation}
The relative residual is thus known at every step at a cost of
$O(k L^3)$, which does not depend on $N$.

\paragraph{Stopping test and solution.}
The iteration stops when the relative residual
\eqref{eq:gmres-residual} falls below the tolerance $\tau$ or when
$n$ reaches $n_{\max}$.
The block $\mathbf{Y}^{(n)}$ is then obtained by back substitution
with the triangular factor, and
$\mathbf{X}^{(n)} = \mathbf{Q}_n \mathbf{Y}^{(n)}$ is assembled.
The defaults are $\tau = 10^{-2}$ and $n_{\max} = 25$.
They are the attributes \texttt{tol} and \texttt{itermax} of
\texttt{DiscreteDipoles}.
The attribute \texttt{converge} is set to \texttt{True} only when the
test is passed.

\paragraph{Cost and memory.}
Each step needs one block matrix--vector product.
The Gram--Schmidt process \eqref{eq:block-gram-schmidt} of step $k$
needs $k$ block inner products and updates of $O(N L^2)$ operations
each.
Its total cost grows quadratically with the number of steps
(Eq.~\eqref{eq:gmres-cost}).
The method is not restarted, so all blocks are kept.
The basis needs $16 \cdot 3N \cdot L \cdot (n+1)$ bytes after $n$
steps.
The triangular factor is allocated for $n_{\max}$ steps and takes
$16\,(n_{\max} L)^2$ bytes.

%----------------------------------------------------------------------
\subsection{Alternative solvers}
\label{sec:alt-solvers}
%----------------------------------------------------------------------

The module \texttt{bl\_krylov/bl\_krylov.py} keeps three other block
solvers.
\texttt{DiscreteDipoles} does not call them.
They take the same arguments as \texttt{bl\_gmres\_mvp\_fft}.
All three use short recurrences, so their memory does not grow with the
number of iterations.

\subsubsection{Jacobi preconditioning}
\label{sec:jacobi}

Two of the alternative solvers apply a left Jacobi (diagonal)
preconditioner:
\begin{equation}
  \mathbf{M}^{-1} = \operatorname{diag}(\mathbf{A})^{-1}
                   = \operatorname{diag}(\alpha_{c,i}),
  \label{eq:jacobi-precond}
\end{equation}
so that the preconditioned system becomes
$\mathbf{M}^{-1}\mathbf{A}\,\mathbf{X} = \mathbf{M}^{-1}\mathbf{B}$.
Their stopping test uses the residual of the preconditioned system.

\subsubsection{Block-BiCGSTAB}
\label{sec:bicgstab}

Block-BiCGSTAB~\cite{Tadano2009} extends the stabilised bi-conjugate
gradient method to block systems.
It is suitable for general (non-symmetric) matrices.
Each iteration requires two block MVPs (lines~1 and~4 of
Table~\ref{tab:bicgstab}).
The function \texttt{bl\_bicgstab\_jacobi\_mvp\_fft} implements the
algorithm of Table~\ref{tab:bicgstab}.
The function \texttt{bl\_bicgstab\_mvp\_fft} is the same algorithm

\subsubsection{Block-COCG-rq}
\label{sec:cocg}

The DDA interaction matrix $\mathbf{A}$ is complex symmetric
($\mathbf{A} = \mathbf{A}^\top$).
Block-COCG-rq~\cite{Gu2016} is a solver specialised for this
structure.
It requires only one block MVP per iteration and is therefore faster
per iteration than Block-BiCGSTAB.
For particles with a high refractive index its convergence is less
stable than that of Block-BiCGSTAB.
The function \texttt{bl\_cocg\_rq\_jacobi\_mvp\_fft} implements it with
Jacobi preconditioning.

%======================================================================
\section{Shifted Block GMRES for Refractive-Index Sweeps}
\label{sec:shifted-gmres}
%======================================================================

A lookup table of scattering observables requires the solution of
\eqref{eq:block-system} for many values of the particle refractive
index~$m_p$ at a fixed shape, size and wavelength.
For a homogeneous isotropic particle these systems differ only by a
scalar multiple of the identity matrix~\cite{Argentin2026}.
This section describes how block-DDA\_Py solves all of them with a
single block-Krylov process.
The method combines the shifted-Krylov idea of
Ref.~\cite{Argentin2026} with the multi-orientation block GMRES of
Ref.~\cite{Moteki2026}.
It is implemented in \texttt{bl\_gmres\_shifted\_mvp\_fft}
(\texttt{bl\_krylov/bl\_krylov.py}) and is available from v0.8.0.

%----------------------------------------------------------------------
\subsection{Shift structure of the interaction matrix}
\label{sec:shift-structure}
%----------------------------------------------------------------------

Split the interaction matrix \eqref{eq:A-matrix} into its diagonal and
off-diagonal parts:
\begin{equation}
  \mathbf{A} = \mathbf{D} + \mathbf{A}_{\text{off}},
  \qquad
  \mathbf{D} = \operatorname{diag}\bigl(\boldsymbol{\alpha}_i^{-1}\bigr).
  \label{eq:A-split}
\end{equation}
The off-diagonal part contains only the Green's tensor.
It depends on the lattice, on $\lambda_0$ and on $m_m$.
It does not depend on $m_p$.
The same holds for the right-hand side $\mathbf{B}$, which holds the
incident field at the lattice points.

For a homogeneous isotropic particle the polarisability
\eqref{eq:rr-polarisability} is one scalar $\alpha(m_p)$ for all
dipoles and all Cartesian components.
Let $m_1, \ldots, m_{N_m}$ be the refractive indices of the sweep.
The system for $m_s$ is
\begin{equation}
  \bigl(\sigma_s\,\mathbf{I} + \mathbf{A}_{\text{off}}\bigr)\,\mathbf{X}_s
    = \mathbf{B},
  \qquad
  \sigma_s = \frac{1}{\alpha(m_s)},
  \qquad
  s = 1, \ldots, N_m.
  \label{eq:shifted-systems}
\end{equation}
In this section $\sigma_s$ denotes a shift, not a cross section.
The formulation \eqref{eq:dda-system} is therefore already in shifted
form and needs no reformulation.

Three conditions must hold.
\begin{enumerate}
  \item The particle is homogeneous and isotropic.
        For a uniaxial particle $\mathbf{D}$ holds two different values.
        A change of $m_p$ then changes $\mathbf{D}$ by a matrix that is
        not a multiple of $\mathbf{I}$.
  \item All $m_s$ share one lattice.
        The lattice spacing is normally set from the refractive index as
        $d = \lambda_0 / (\max_c |m_{p,c}|\, n_{\text{dpl}})$, where
        $n_{\text{dpl}}$ is the number of dipoles per wavelength inside
        the particle.
        In a shifted sweep the lattice is built for $\max_s |m_s|$.
        The lower refractive indices are then discretised more finely
        than the rule requires.
  \item No preconditioner is applied.
        The solver is the unpreconditioned block GMRES.
\end{enumerate}

%----------------------------------------------------------------------
\subsection{Block Arnoldi process and shift invariance}
\label{sec:shift-arnoldi}
%----------------------------------------------------------------------

Let $\mathbf{B} = \mathbf{V}_1 \boldsymbol{\Lambda}$ be the thin QR
factorisation of the right-hand side, with
$\mathbf{V}_1 \in \mathbb{C}^{3N \times L}$ and
$\boldsymbol{\Lambda} \in \mathbb{C}^{L \times L}$.
The block Arnoldi process applied to $\mathbf{A}_{\text{off}}$ generates
blocks $\mathbf{V}_1, \mathbf{V}_2, \ldots$ with mutually orthonormal
columns.
After $n$ steps it satisfies
\begin{equation}
  \mathbf{A}_{\text{off}}\,\mathbf{Q}_n
    = \mathbf{Q}_{n+1}\,\bar{\mathbf{H}}_n,
  \qquad
  \mathbf{Q}_n = [\mathbf{V}_1, \ldots, \mathbf{V}_n],
  \label{eq:block-arnoldi}
\end{equation}
where $\bar{\mathbf{H}}_n \in \mathbb{C}^{(n+1)L \times nL}$ is block
upper Hessenberg.
Adding $\sigma_s \mathbf{Q}_n$ to both sides gives
\begin{equation}
  \bigl(\sigma_s\,\mathbf{I} + \mathbf{A}_{\text{off}}\bigr)\,\mathbf{Q}_n
    = \mathbf{Q}_{n+1}\,
      \bigl(\bar{\mathbf{H}}_n + \sigma_s\,\bar{\mathbf{I}}_n\bigr),
  \qquad
  \bar{\mathbf{I}}_n =
  \begin{bmatrix} \mathbf{I}_{nL} \\ \mathbf{0}_{L \times nL} \end{bmatrix}.
  \label{eq:shifted-arnoldi}
\end{equation}
The block Krylov subspace spanned by $\mathbf{Q}_n$ is thus the same for
every shift~\cite{FrommerGlassner1998}.
One Arnoldi process serves all $N_m$ systems.
Each step costs one block matrix--vector product with
$\mathbf{A}_{\text{off}}$ (Sec.~\ref{sec:goodman-fft}) and one block
Gram--Schmidt orthogonalisation against the previous blocks.

%----------------------------------------------------------------------
\subsection{GMRES iterate of each shift}
\label{sec:shift-iterate}
%----------------------------------------------------------------------

With the initial guess $\mathbf{X}_s = \mathbf{0}$, the block GMRES
iterate after $n$ steps is sought in the form
$\mathbf{X}_s^{(n)} = \mathbf{Q}_n \mathbf{Y}$ with
$\mathbf{Y} \in \mathbb{C}^{nL \times L}$.
Because $\mathbf{Q}_{n+1}$ has orthonormal columns and
$\mathbf{B} = \mathbf{Q}_{n+1} \mathbf{E}_1 \boldsymbol{\Lambda}$ with
$\mathbf{E}_1 = [\mathbf{I}_L, \mathbf{0}, \ldots, \mathbf{0}]^\top$,
\eqref{eq:shifted-arnoldi} gives
\begin{equation}
  \bigl\| \mathbf{B}
      - (\sigma_s\,\mathbf{I} + \mathbf{A}_{\text{off}})\,
        \mathbf{Q}_n \mathbf{Y} \bigr\|_F
  = \bigl\| \mathbf{E}_1 \boldsymbol{\Lambda}
      - (\bar{\mathbf{H}}_n + \sigma_s\,\bar{\mathbf{I}}_n)\,
        \mathbf{Y} \bigr\|_F .
  \label{eq:shifted-residual}
\end{equation}
The iterate of shift $s$ is the minimiser
\begin{equation}
  \mathbf{Y}_s^{(n)}
    = \arg\min_{\mathbf{Y}}
      \bigl\| \mathbf{E}_1 \boldsymbol{\Lambda}
        - (\bar{\mathbf{H}}_n + \sigma_s\,\bar{\mathbf{I}}_n)\,
          \mathbf{Y} \bigr\|_F ,
  \qquad
  \mathbf{X}_s^{(n)} = \mathbf{Q}_n \mathbf{Y}_s^{(n)} .
  \label{eq:shifted-lsq}
\end{equation}
Unrestarted block GMRES (Sec.~\ref{sec:block-gmres}) applied to
system $s$ alone minimises the same residual over the same subspace.
The iterate \eqref{eq:shifted-lsq} is therefore identical to the one of
an independent solve.
The stopping test is applied to each shift separately:
\begin{equation}
  \frac{\bigl\| \mathbf{B}
      - (\sigma_s\,\mathbf{I} + \mathbf{A}_{\text{off}})\,
        \mathbf{X}_s^{(n)} \bigr\|_F}{\| \mathbf{B} \|_F}
  < \text{tol}.
  \label{eq:shifted-stop}
\end{equation}
A shift that passes the test at step $n_s$ keeps its
$\mathbf{Y}_s^{(n_s)}$ and is dropped from the later steps.
The process ends when every shift has stopped or when the maximum
number of steps is reached.
The number of steps $n_s$ equals that of an independent solve of
system~$s$.
The Arnoldi process runs for $\max_s n_s$ steps.

The least-squares problem \eqref{eq:shifted-lsq} is solved by a QR
factorisation of the augmented matrix:
\begin{equation}
  \bigl[\, \bar{\mathbf{H}}_n + \sigma_s\,\bar{\mathbf{I}}_n \;\big|\;
         \mathbf{E}_1 \boldsymbol{\Lambda} \,\bigr]
  = \tilde{\mathbf{Q}}
    \begin{bmatrix}
      \mathbf{R} & \mathbf{C} \\
      \mathbf{0} & \mathbf{R}_{\text{res}}
    \end{bmatrix},
  \qquad
  \mathbf{Y}_s^{(n)} = \mathbf{R}^{-1} \mathbf{C},
  \qquad
  \text{residual norm} = \| \mathbf{R}_{\text{res}} \|_F .
  \label{eq:augmented-qr}
\end{equation}
The matrix $\bar{\mathbf{H}}_n + \sigma_s\,\bar{\mathbf{I}}_n$ is block
upper Hessenberg, so the factorisation \eqref{eq:augmented-qr} does not
have to be recomputed at every step.
The residual norm of every active shift is tracked by the incremental
block QR of Sec.~\ref{sec:block-gmres}.
Step $k$ carries the new block column of
$\bar{\mathbf{H}}_k + \sigma_s\,\bar{\mathbf{I}}_k$ through the
transformations of the steps $1, \ldots, k-1$ of shift $s$.
The shift enters the diagonal block of this column only.
A QR factorisation of a $2L \times L$ block then gives the
transformation of step $k$, which is applied to the right-hand side.
This costs $O(k L^3)$ per shift and step.
The $2L \times 2L$ transformations of each shift are kept.
They take $64\,n L^2$ bytes per shift after $n$ steps.
The triangular factor $\mathbf{R}$ is not stored.
When shift $s$ stops, $\mathbf{Y}_s^{(n_s)}$ is computed once.
The stored transformations of shift $s$ are applied to the augmented
matrix of \eqref{eq:augmented-qr}, which is built from the shared
matrix $\bar{\mathbf{H}}_{n_s}$.
This costs $O(n_s^2 L^3)$.
None of these costs depends on $N$.

The option \texttt{lsq\_method="dense"} of
\texttt{bl\_gmres\_shifted\_mvp\_fft} selects the method of v0.8.0
instead.
The incremental method is the default from v0.8.1.
It computes \eqref{eq:augmented-qr} from scratch for every active shift
at every step, without using the Hessenberg structure.
This keeps no state per shift but costs $O\bigl((kL)^3\bigr)$ per shift
and step.
The two methods give the same step counts and the same iterates up to
rounding error.

The shift invariance holds for the unrestarted method only.
After a restart the residuals of different shifts span different
spaces, and a common basis no longer serves all
shifts~\cite{FrommerGlassner1998}.
block-DDA\_Py keeps the full Krylov basis, so this limitation applies
only when the basis no longer fits in memory.

%----------------------------------------------------------------------
\subsection{Observables without assembling the solutions}
\label{sec:shift-observables}
%----------------------------------------------------------------------

Assembling $\mathbf{X}_s = \mathbf{Q}_n \mathbf{Y}_s$ costs
$O(3N \cdot nL \cdot L)$ operations per shift.
For $N_m$ shifts this is comparable to the Gram--Schmidt cost of the
whole Arnoldi process.
The observables of Sec.~\ref{sec:observables} are linear or quadratic
functionals of the solution and can be evaluated from
$\mathbf{Y}_s$ directly.
Let $\bm{x}_{s,\ell}$ and $\bm{y}_{s,\ell}$ denote column $\ell$ of
$\mathbf{X}_s$ and $\mathbf{Y}_s$, and $\bm{b}_\ell$ column $\ell$ of
$\mathbf{B}$.

\paragraph{Extinction.}
Equation \eqref{eq:C-ext} reads
$C_{\text{ext},\ell} = 4\pi k \operatorname{Im}(\bm{b}_\ell^\dagger \bm{x}_{s,\ell})$.
Since
$\mathbf{B}^\dagger \mathbf{Q}_n
  = [\boldsymbol{\Lambda}^\dagger, \mathbf{0}, \ldots, \mathbf{0}]$,
\begin{equation}
  C_{\text{ext},\ell}(m_s)
    = 4\pi k \operatorname{Im}
      \bigl[ \bigl( \boldsymbol{\Lambda}^\dagger\,
             \mathbf{Y}_{s,1} \bigr)_{\ell\ell} \bigr],
  \label{eq:C-ext-shifted}
\end{equation}
where $\mathbf{Y}_{s,1}$ is the first $L$ rows of $\mathbf{Y}_s$.

\paragraph{Scattering amplitudes.}
Each amplitude in \eqref{eq:S-fw-theta}, \eqref{eq:S-fw-phi} and their
backward counterparts has the form
$S_\ell = \bm{f}_\ell^\dagger \bm{x}_{s,\ell}$.
The vector $\bm{f}_\ell \in \mathbb{C}^{3N}$ has the entries
\begin{equation}
  (\bm{f}_\ell)_{i,c} = k^2\, \kappa\, e_i\, (\hat{\bm{e}})_c ,
  \label{eq:functional-f}
\end{equation}
where $e_i$ is the phase factor of dipole $i$ for the scattering
direction, $\hat{\bm{e}}$ is the real polarisation unit vector, and
$\kappa = \sqrt{2}$ for the $\theta$-component and
$\kappa = \sqrt{2}\,i$ for the $\phi$-component.
Collect the vectors as
$\mathbf{F} = [\bm{f}_1, \ldots, \bm{f}_L]$ and accumulate
\begin{equation}
  \mathbf{G}_n = \mathbf{F}^\dagger \mathbf{Q}_n
  \label{eq:projection-G}
\end{equation}
during the Arnoldi process, one $L \times L$ block per step.
Then
\begin{equation}
  S_\ell(m_s) = \bigl( \mathbf{G}_{n_s} \mathbf{Y}_s \bigr)_{\ell\ell} .
  \label{eq:S-shifted}
\end{equation}

\paragraph{Absorption.}
By \eqref{eq:E-from-P} the internal field is
$\bm{E}_i = c_E\, \bm{P}_i$ with
$c_E(m) = 4\pi / [(\epsilon_r - 1)\, v]$.
Equation \eqref{eq:C-abs} then becomes
$C_{\text{abs},\ell} = 4\pi k \operatorname{Im}(c_E^*)\, \| \bm{x}_{s,\ell} \|^2$.
Because $\mathbf{Q}_n$ has orthonormal columns,
$\| \bm{x}_{s,\ell} \| = \| \bm{y}_{s,\ell} \|$ and
\begin{equation}
  C_{\text{abs},\ell}(m_s)
    = 4\pi k \operatorname{Im}\bigl[ c_E(m_s)^* \bigr]\,
      \| \bm{y}_{s,\ell} \|^2 .
  \label{eq:C-abs-shifted}
\end{equation}

\paragraph{Forward amplitudes and extinction.}
In the forward direction the phase factor $e_i$ equals that of the
incident wave.
With the circular polarisation \eqref{eq:circular-pol}, the two forward
functionals add up to the right-hand side:
$\bm{f}_{\theta,\ell} + \bm{f}_{\phi,\ell} = 2k^2\, \bm{b}_\ell$.
Hence
\begin{equation}
  C_{\text{ext},\ell}
    = \frac{2\pi}{k} \operatorname{Im}
      \bigl( S_{\text{fw},\theta}^{\text{PCAS}}
           + S_{\text{fw},\phi}^{\text{PCAS}} \bigr)_\ell ,
  \label{eq:optical-theorem}
\end{equation}
which is the optical theorem in the present normalisation.
It holds for the existing solver as well and is used as a consistency
check.

%----------------------------------------------------------------------
\subsection{Common-offset sweep of an anisotropic particle}
\label{sec:shift-offset}
%----------------------------------------------------------------------

The diagonal of $\mathbf{A}$ of an anisotropic particle holds three
different values $1/\alpha_c$ (Sec.~\ref{sec:polarisability}).
A change of the refractive indices then changes $\mathbf{A}$ by a
diagonal matrix that is in general not a scalar multiple of the
identity.
One family of sweeps keeps the shift structure.
Equations~\eqref{eq:cm-polarisability} and \eqref{eq:rr-polarisability}
give
\begin{equation}
  \frac{1}{\alpha_c}
    = \frac{4\pi}{3v}\,\bigl(1 + 3\,\tau_c\bigr) - \frac{M}{v},
  \qquad
  \tau_c = \frac{1}{\epsilon_{r,c} - 1} .
  \label{eq:inv-alpha-tau}
\end{equation}
The inverse polarisability depends on the refractive index only
through $\tau_c$ and is linear in it.
Let $\tau_c^{\text{ref}}$ be the values at a reference point with the
principal refractive indices $m_{p,c}^{\text{ref}}$.
A sweep that adds the same offset to all three axes,
\begin{equation}
  \tau_c^{(s)} = \tau_c^{\text{ref}} + \Delta\tau_s ,
  \qquad c \in \{x, y, z\},
  \label{eq:common-offset}
\end{equation}
gives the family
\begin{equation}
  \mathbf{A}_s = \mathbf{A}_{\text{ref}} + \sigma_s\,\mathbf{I},
  \qquad
  \sigma_s = \frac{4\pi}{v}\,\Delta\tau_s,
  \qquad
  \mathbf{A}_{\text{ref}}
    = \operatorname{diag}\bigl(1/\alpha_c^{\text{ref}}\bigr)
      + \mathbf{A}_{\text{off}} .
  \label{eq:offset-family}
\end{equation}
The block Arnoldi process is applied to $\mathbf{A}_{\text{ref}}$
instead of $\mathbf{A}_{\text{off}}$.
Sections~\ref{sec:shift-arnoldi} and~\ref{sec:shift-iterate} then hold
without change.
The principal refractive indices at sweep point $s$ are
\begin{equation}
  m_{p,c}^{(s)} = m_m \sqrt{1 + \frac{1}{\tau_c^{(s)}}} .
  \label{eq:offset-indices}
\end{equation}
The sweep is parametrised by the refractive index of the $x$ axis.
For a uniaxial particle with $m_p = (n_o, n_o, n_e)$ this is the
ordinary index $n_o$.

The quantity that stays constant along the sweep is
$\tau_e - \tau_o$.
The birefringence $n_e - n_o$ is not constant.
It grows with the index contrast and vanishes where $n_o$ approaches
$m_m$.
Table~\ref{tab:offset-birefringence} shows an example.
A sweep at fixed $n_e - n_o$ does not have the form
\eqref{eq:offset-family} and needs one solve per refractive index.

\begin{table}[htbp]
  \centering
  \caption{Birefringence $n_e - n_o$ along a common-offset sweep whose
           reference point is $n_o = 1.50$, $n_e = 1.55$.}
  \label{tab:offset-birefringence}
  \begin{tabular}{lrrrrr}
    \toprule
    $n_o$ & 1.35 & 1.40 & 1.50 & 1.60 & 1.70 \\
    \midrule
    Air ($m_m = 1$)        & 0.023  & 0.031  & 0.050 & 0.075 & 0.106 \\
    Water ($m_m = 1.3345$) & 0.0003 & 0.0066 & 0.050 & 0.159 & 0.403 \\
    \bottomrule
  \end{tabular}
\end{table}

The method \texttt{set\_interaction\_matrix\_shifted\_offset} of the
scatterer class sets up the sweep.
The reference point is the \texttt{m\_p\_xyz} of the object, and the
method returns the indices \eqref{eq:offset-indices} of every sweep
point.
The solver receives the diagonal of $\mathbf{A}_{\text{ref}}$ through
the argument \texttt{diag\_ref} of
\texttt{bl\_gmres\_\allowbreak shifted\_\allowbreak mvp\_fft}.
The lattice must be built for the largest $|m_{p,c}^{(s)}|$ of the
sweep.

The extinction cross section and the scattering amplitudes are
obtained from $\mathbf{Y}_s$ as in Sec.~\ref{sec:shift-observables}.
The absorption cross section \eqref{eq:C-abs-shifted} is not, because
its prefactor differs between the three axes.
For an absorbing anisotropic particle the solutions $\mathbf{X}_s$ are
assembled and $C_{\text{abs}}$ is evaluated as in
Sec.~\ref{sec:observables}.

The tests compare the sweep with independent solves of the existing
anisotropic path on the same lattice.
The case is the spheroid of Sec.~\ref{sec:shift-verification} with the
reference $m_p = (1.60, 1.60, 1.68)$ and the ordinary indices
$n_o = 1.40$, $1.50$, $1.60$ and $1.50 + 0.05i$.
The step counts are identical ($4, 5, 6, 5$ at $\text{tol} = 10^{-2}$
and $9, 11, 12, 11$ at $\text{tol} = 10^{-5}$).
The solutions differ by at most \num{7.1e-16} and the observables by at
most \num{3.4e-15} in relative terms.

The sweep driver does not use the common-offset sweep.
Its option \texttt{--delta-n-eff} keeps $n_e - n_o$ fixed along the
$n_{\mathrm{real}}$ axis and solves each refractive index separately.

%----------------------------------------------------------------------
\subsection{Cost and memory}
\label{sec:shift-cost}
%----------------------------------------------------------------------

Let $t_{\text{mvp}}$ be the time of one block matrix--vector product and
$t_{\text{gs}}$ the time of one block inner product and update in the
Gram--Schmidt process.
A block GMRES run of $K$ steps costs
\begin{equation}
  T(K) \approx K\, t_{\text{mvp}} + \frac{K(K+1)}{2}\, t_{\text{gs}} .
  \label{eq:gmres-cost}
\end{equation}
The independent solves of the sweep cost $\sum_s T(n_s)$ and the
shifted solve costs $T(\max_s n_s)$ plus the small least-squares
problems.
The solver speed-up is therefore
\begin{equation}
  S_{\text{solve}} \approx \frac{\sum_{s=1}^{N_m} T(n_s)}{T(\max_s n_s)}
    \le N_m .
  \label{eq:shift-speedup}
\end{equation}
It approaches $N_m$ when all refractive indices converge at a similar
rate.
The lattice generation and the FFT of the interaction tensor are done
once instead of $N_m$ times.

The Krylov basis needs $16 \cdot 3N \cdot L \cdot (n+1)$ bytes, the
same as one independent solve on the same lattice.
The additional storage is $\bar{\mathbf{H}}_n$ and the blocks
$\mathbf{Y}_s$, which do not scale with $N$.
The number of dipoles on the common lattice scales as
$\max_s |m_s|^3$.
A sweep over a wide range of $|m_p|$ should therefore be split into
bands, each with its own lattice.

%----------------------------------------------------------------------
\subsection{Use in the spheroid parameter sweep}
\label{sec:shift-sweep}
%----------------------------------------------------------------------

The option \texttt{--shifted-ri} of
\texttt{run\_dda\_spheroid\_sweep.py} (Sec.~\ref{sec:spheroid-sweep})
solves all $N_{\mathrm{RI}}$ values of $n_{\mathrm{real}}$ of a
$(\lambda_0, D_{ve}, \mathrm{AR})$ slot with one shifted solve.
The number of solves per wavelength drops from
$N_{D_{ve}} \times N_{\mathrm{RI}} \times N_{\mathrm{AR}}$ to
$N_{D_{ve}} \times N_{\mathrm{AR}}$.
The lattice of a slot is always built for the largest $|m_p|$ of the
whole $n_{\mathrm{real}}$ axis.
A run that is interrupted and resumed therefore reproduces an
uninterrupted run.

The default mode builds one lattice per refractive index.
The two modes agree to rounding error at the largest
$n_{\mathrm{real}}$, where the lattices coincide.
At the lower values they differ at the level of the discretisation
error.
An output file keeps the mode it was created with.
The root attributes \texttt{shifted\_ri} and \texttt{lattice\_policy}
mark a file of the shifted mode.
The option cannot be combined with a nonzero uniaxial birefringence
(\texttt{--delta-n-eff}).

\subsection{Verification}
\label{sec:shift-verification}
%----------------------------------------------------------------------

The shifted path is compared with the existing solver in the pytest
suite under \texttt{tests/}.
The reference case is an oblate spheroid with
$D_{ve} = \SI{0.35}{\micro\meter}$ and $b/c = 2$ at
$\lambda_0 = \SI{0.638}{\micro\meter}$ in air, with $L = 5$

\section{Computational Performance: Old vs.\ New Implementation}
\label{sec:performance}
%======================================================================

This section compares the computational cost of the previous
implementation (Barrowes 2001 block-Toeplitz FFT with \texttt{ray}
multiprocessing) and the current implementation (Goodman 1991 3D FFT
with NumPy/SciPy broadcasting).
The estimates were made when Block-BiCGSTAB was the solver, with two
block MVPs per iteration.
The comparison per block MVP applies to block GMRES as well.

%----------------------------------------------------------------------
\subsection{MVP complexity}
\label{sec:mvp-complexity}
%----------------------------------------------------------------------

Both algorithms compute the off-diagonal matrix--vector product via
FFT-based circular convolution, and both have the same asymptotic
complexity per right-hand side (RHS):
\[
  T_{\text{MVP}}^{(1)} = O(N \log N),
\]
where $N = N_x N_y N_z$ is the total cuboid grid size.  For $L$
right-hand sides the total floating-point operation count is
$O(L N \log N)$ in both cases.  The difference lies in execution
efficiency.

\label{sec:parallel-overhead}

\section{Scattering Observables}
\label{sec:observables}
%======================================================================

Once the polarisation $\bm{P}_i$ ($i = 1, \ldots, N$) has been obtained
for each orientation~$\ell$, the following observables are computed.
Throughout this section, $k$ denotes the wavenumber in the medium as
defined in \eqref{eq:wavenumber}.

%----------------------------------------------------------------------
\subsection{Absorption and extinction cross sections}
\label{sec:cross-sections}
%----------------------------------------------------------------------

The absorption cross section is~\cite{Draine1988}
\begin{equation}
  C_{\text{abs}}
    = 4\pi k \sum_{i=1}^{N}
      \operatorname{Im}\!\bigl(\bm{P}_i \cdot \bm{E}_i^*\bigr),
  \label{eq:C-abs}
\end{equation}
where $\bm{E}_i$ is the internal electric field at dipole~$i$,
reconstructed from the polarisation via
\begin{equation}
  \bm{E}_i
    = \frac{4\pi\,\bm{P}_i}{(\epsilon_{r,i} - 1)\,v}.
  \label{eq:E-from-P}
\end{equation}

The extinction cross section is
\begin{equation}
  C_{\text{ext}}
    = 4\pi k \sum_{i=1}^{N}
      \operatorname{Im}\!\bigl(\bm{P}_i \cdot \bm{E}_{\text{inc},i}^*\bigr).
  \label{eq:C-ext}
\end{equation}

The corresponding efficiency factors are
$Q_{\text{abs}} = C_{\text{abs}} / (\pi r_{ve}^2)$ and
$Q_{\text{ext}} = C_{\text{ext}} / (\pi r_{ve}^2)$,
where $r_{ve}$ is the volume-equivalent sphere radius:
\begin{equation}
  r_{ve} = \left(\frac{3\,N\,v}{4\pi}\right)^{\!1/3}.
  \label{eq:r-ve}
\end{equation}

%----------------------------------------------------------------------
\subsection{Forward scattering amplitude (PCAS)}
\label{sec:S-fw}
%----------------------------------------------------------------------

The forward scattering amplitude in the PCAS (Polarimetric Complex
Amplitude of Scattering) framework is computed from the far-field
projection of the dipole polarisations.

For the forward scattering direction $\hat{\bm{u}}_{\text{fw}} = +\hat{\bm{z}}$
in the laboratory frame, define the phase factor for each dipole:
\begin{equation}
  e_{\text{fw},i}
    = \exp\!\bigl(ik\,\hat{\bm{u}}_{\text{fw}} \cdot \bm{r}_i\bigr).
  \label{eq:e-fw}
\end{equation}

The $\theta$- and $\phi$-components of the forward scattering amplitude
correspond to the s- and p-polarisation channels of the CAS-v2
detector, respectively:
$S_{\text{fw},\theta}^{\text{PCAS}} \equiv S_s(0)$ and
$S_{\text{fw},\phi}^{\text{PCAS}} \equiv S_p(0)$.
They are computed as
\begin{align}
  S_{\text{fw},\theta}^{\text{PCAS}}\; [= S_s(0)]
    &= k^2 \sum_{i=1}^{N}
       (\bm{P}_i \cdot \hat{\bm{e}}_{\theta,\text{fw}})\;
       (\sqrt{2}\; e_{\text{fw},i})^*,
  \label{eq:S-fw-theta} \\
  S_{\text{fw},\phi}^{\text{PCAS}}\; [= S_p(0)]
    &= k^2 \sum_{i=1}^{N}
       (\bm{P}_i \cdot \hat{\bm{e}}_{\phi,\text{fw}})\;
       (\sqrt{2}\,i\; e_{\text{fw},i})^*,
  \label{eq:S-fw-phi}
\end{align}
where the factors $\sqrt{2}$ and $\sqrt{2}\,i$ arise from the
circular-polarisation decomposition of the incident field
\eqref{eq:circular-pol}.

%----------------------------------------------------------------------
\subsection{Backward scattering amplitude (OCBS)}
\label{sec:S-bk}
%----------------------------------------------------------------------

For the exact backward scattering direction
$\hat{\bm{u}}_{\text{bk}} = -\hat{\bm{z}}$, define the phase factor
\begin{equation}
  e_{\text{bk},i}
    = \exp\!\bigl(ik\,\hat{\bm{u}}_{\text{bk}} \cdot \bm{r}_i\bigr).
  \label{eq:e-bk}
\end{equation}
Let $\hat{\bm{e}}_{\theta,\text{bk}}$ and
$\hat{\bm{e}}_{\phi,\text{bk}}$ be the polar and azimuthal unit
vectors associated with the backward direction.
In the laboratory frame:
$\hat{\bm{e}}_{\theta,\text{bk}} = -\hat{\bm{x}}$ and
$\hat{\bm{e}}_{\phi,\text{bk}} = +\hat{\bm{y}}$.

The $\theta$- and $\phi$-components of the backward scattering
amplitude are
\begin{align}
  S_{\text{bk},\theta}^{\text{OCBS}}
    &= k^2 \sum_{i=1}^{N}
       (\bm{P}_i \cdot \hat{\bm{e}}_{\theta,\text{bk}})\;
       (\sqrt{2}\; e_{\text{bk},i})^*,
  \label{eq:S-bk-theta} \\
  S_{\text{bk},\phi}^{\text{OCBS}}
    &= k^2 \sum_{i=1}^{N}
       (\bm{P}_i \cdot \hat{\bm{e}}_{\phi,\text{bk}})\;
       (\sqrt{2}\,i\; e_{\text{bk},i})^*,
  \label{eq:S-bk-phi}
\end{align}
where the reference-wave factors $\sqrt{2}$ and $\sqrt{2}\,i$ are the
same as in the forward case.

The OCBS (Optical Coherence Backscattering Sensor) amplitude is then
\begin{equation}
  S_{\text{bk}}^{\text{OCBS}}
    = \frac{-S_{\text{bk},\theta}^{\text{OCBS}}
            + S_{\text{bk},\phi}^{\text{OCBS}}}{\sqrt{2}}.
  \label{eq:S-bk-ocbs}
\end{equation}
More precisely, the $\theta$- and $\phi$-components correspond to
combinations of the Mishchenko amplitude matrix
elements~\cite{Mishchenko2002}:
\begin{equation}
  S_{\text{bk},\theta}^{\text{OCBS}}
    \equiv S_{11}(\pi) + i\,S_{12}(\pi),
  \qquad
  S_{\text{bk},\phi}^{\text{OCBS}}
    \equiv S_{22}(\pi) - i\,S_{21}(\pi).
  \label{eq:S-bk-mishchenko}
\end{equation}
The backscattering theorem $S_{21}(\pi) = -S_{12}(\pi)$ causes the
off-diagonal contributions to cancel when the two components are
combined in \eqref{eq:S-bk-ocbs}, yielding an expression equivalent to
Eq.~(28) of Moteki and Adachi~\cite{Moteki2024}.

\section{Gaussian Random Ellipsoid (GRE) Shape Model}
\label{sec:gre}
%======================================================================

The particle shape is generated using the Gaussian Random Ellipsoid
(GRE) model of Muinonen \& Pieniluoma~\cite{Muinonen2011}.

%----------------------------------------------------------------------
\subsection{Base ellipsoid}
\label{sec:base-ellipsoid}
%----------------------------------------------------------------------

The base ellipsoid is defined by three semi-axes $a \ge b \ge c$ and
the volume-equivalent radius $r_v$.
Two axis ratios are specified as input: $b/c$ (\texttt{bc\_ratio}) and
$a/b$ (\texttt{ab\_ratio}).
The semi-axis~$c$ is determined by volume conservation:
\begin{equation}
  c = \left(\frac{r_v^3}{(a/b)\,(b/c)^2}\right)^{\!1/3},
  \qquad
  b = c \cdot (b/c),
  \qquad
  a = b \cdot (a/b).
  \label{eq:semi-axes}
\end{equation}

%----------------------------------------------------------------------
\subsection{Surface deformation}
\label{sec:gre-deformation}
%----------------------------------------------------------------------

A Gaussian random field~$s(\theta, \phi)$ is sampled on the surface of
the base ellipsoid with covariance
\begin{equation}
  \operatorname{Cov}(s_i, s_j)
    = \beta^2 \exp\!\left(
        -\frac{\|\bm{r}_i - \bm{r}_j\|^2}{2\,\ell_c^2}
      \right),
  \label{eq:gre-covariance}
\end{equation}
where $\beta$ is the standard deviation of the deformation and
$\ell_c$ is the correlation length, which controls the fineness of
the surface roughness.
In the current version (v0.6.0), $\ell_c = 0.3\,c$ is adopted as an
empirical choice.

The deformed surface position is obtained by displacing each point
on the base ellipsoid along the outward surface normal
$\hat{\bm{n}}_0$:
\begin{equation}
  \bm{r}_{\text{GRE}}(\theta, \phi)
    = \bm{r}_{\text{base}}(\theta, \phi)
      + h_0 \bigl[\exp\!\bigl(s(\theta,\phi)\bigr)
        - \tfrac{1}{2}\beta^2 - 1\bigr]\;
      \hat{\bm{n}}_0(\theta, \phi),
  \label{eq:gre-deformation}
\end{equation}
where $h_0 = c^2 / a$ is a length scale and $\hat{\bm{n}}_0$ is the
unit outward normal to the base ellipsoid.

%----------------------------------------------------------------------
\subsection{Lattice generation and interior detection}
\label{sec:interior-detection}
%----------------------------------------------------------------------

The lattice spacing is determined by the number of dipoles per
wavelength inside the particle (dpl):
\begin{equation}
  d = \frac{\lambda_{\text{particle,min}}}{\mathrm{dpl}}
    = \frac{\lambda_0}{\max_c |m_{p,c}| \times \mathrm{dpl}},
  \label{eq:lattice-spacing}
\end{equation}
where $m_{p,c}$ ($c \in \{x,y,z\}$) is the complex particle
refractive index along axis~$c$ (cf.\ Sec.~\ref{sec:polarisability})
and $\lambda_{\text{particle,min}} = \lambda_0 / \max_c|m_{p,c}|$
is the shortest wavelength inside the particle.
The default value is $\mathrm{dpl} = 17$, which is a typical choice
for DDA simulations of sub-micrometre particles.
A cubic lattice with this spacing is generated within the bounding
box of the deformed surface.
Interior points are identified by a three-axis ray-casting algorithm.

%----------------------------------------------------------------------
\subsection{Volume-preserving lattice rescaling}
\label{sec:volume-rescaling}
%----------------------------------------------------------------------

Because the GRE surface is discretised onto a cubic lattice, the
total volume of the $N$ occupied dipoles, $N d^3$, generally differs
from the target volume $V_{\text{target}} = \tfrac{4}{3}\pi r_{v,\text{base}}^3$
by a few percent.  When parameter sweeps require exact control of the
volume-equivalent radius (e.g.\ an equally-spaced $r_{v,\text{base}}$
grid), this discretisation error is undesirable.

To eliminate it, the lattice spacing is rescaled \emph{after} the
interior detection step.  Given the number of occupied sites~$N$ and
the desired $r_{v,\text{base}}$, the adjusted spacing is
\begin{equation}
  d_{\text{adj}}
    = \left(\frac{V_{\text{target}}}{N}\right)^{\!1/3}
    = \left(\frac{4\pi}{3N}\right)^{\!1/3} r_{v,\text{base}}.
  \label{eq:d-adj}
\end{equation}
All lattice coordinates are then multiplied by the scale factor
$s = d_{\text{adj}} / d$, and $d_{\text{adj}}$ replaces $d$ in every
subsequent computation: the Green's function~\eqref{eq:green-tensor},
the Clausius--Mossotti polarisability~\eqref{eq:cm-polarisability},
and the radiative correction~\eqref{eq:rr-polarisability}.
Because both the inter-dipole distances and the dipole volume
$v = d_{\text{adj}}^3$ change by the same factor, the DDA formulation
remains self-consistent.

This rescaling is performed in the \texttt{Target} constructor
(\texttt{bl\_dda/scatterer.py}), which requires \texttt{r\_v\_base}
as a mandatory argument.

\section{Spheroid Symmetry and Orientation Averaging}
\label{sec:spheroid}
%======================================================================

For spheroids (axially symmetric particles with $a = b$ and GRE roughness
$\beta_{\mathrm{GRE}} = 0$), the scattering amplitudes possess analytical
constraint relations that enable significant computational savings in
multi-orientation calculations.

%----------------------------------------------------------------------
\subsection{Euler angle mapping}
\label{sec:spheroid-euler}
%----------------------------------------------------------------------

The intrinsic ZYZ Euler angles $(\alpha,\beta,\gamma)$ relate to the
spheroid geometry as follows:
\begin{itemize}
  \item $\beta$: polar angle of the symmetry axis
        (tilt from the laboratory $z$-axis),
  \item $\alpha$: azimuthal angle of the symmetry axis
        (rotation of the tilt plane about $z$),
  \item $\gamma$: rotation about the symmetry axis;
        irrelevant for a spheroid ($a = b$).
\end{itemize}
In the notation of Sec.~\ref{sec:spheroid-constraint},
$\theta \equiv \beta$ and $\phi \equiv \alpha$.

%----------------------------------------------------------------------
\subsection{Constraint relations at forward scattering}
\label{sec:spheroid-constraint}
%----------------------------------------------------------------------

When the symmetry axis lies in the $xz$-plane ($\alpha = 0$), the
cross-polarisation elements of the amplitude scattering matrix vanish:
$S_{12}(\theta,0) = S_{21}(\theta,0) = 0$.
The forward-scattering amplitudes at arbitrary~$\alpha$ are then
determined entirely by the two diagonal elements at $\alpha = 0$:
\begin{align}
  S_s(\theta,\alpha)
    &= \frac{S_{11}(\theta,0) + S_{22}(\theta,0)}{2}
     + \frac{S_{11}(\theta,0) - S_{22}(\theta,0)}{2}\,e^{i2\alpha},
  \label{eq:Ss-constraint} \\
  S_p(\theta,\alpha)
    &= \frac{S_{11}(\theta,0) + S_{22}(\theta,0)}{2}
     - \frac{S_{11}(\theta,0) - S_{22}(\theta,0)}{2}\,e^{i2\alpha},
  \label{eq:Sp-constraint}
\end{align}
where $S_s \equiv S_{11} + iS_{12}$ and $S_p \equiv S_{22} - iS_{21}$
are the PCAS observables (Sec.~\ref{sec:S-fw}).

These imply three verifiable symmetries:
\begin{enumerate}
  \item \textbf{Trace invariance:}\;
    $S_s(\alpha) + S_p(\alpha)
       = S_{11}(\theta,0) + S_{22}(\theta,0) = \text{const}$.
  \item \textbf{Modulus preservation:}\;
    $|S_s(\alpha) - S_p(\alpha)|
       = |S_{11}(\theta,0) - S_{22}(\theta,0)| = \text{const}$.
  \item \textbf{Cross-symmetry:}\;
    $S_s(\theta,\,\alpha+\pi/2) = S_p(\theta,\,\alpha)$.
\end{enumerate}
These relations are verified numerically at machine precision
($\sim 10^{-16}$) in \texttt{test\_spheroid\_symmetry.ipynb}.

%----------------------------------------------------------------------
\subsection{Analytical phi-averaging (labor-saving mode)}
\label{sec:spheroid-saving}
%----------------------------------------------------------------------

For orientation-averaged observables, the azimuthal average
over $\alpha \in [0,2\pi]$ eliminates the oscillatory term
$e^{i2\alpha}$, yielding
\begin{equation}
  \langle S_s(\theta) \rangle_\alpha
  = \langle S_p(\theta) \rangle_\alpha
  = \frac{S_{11}(\theta,0) + S_{22}(\theta,0)}{2}.
  \label{eq:S-phi-avg}
\end{equation}
Furthermore, $C_{\mathrm{ext}}$ and $C_{\mathrm{abs}}$ are also
$\alpha$-invariant for a spheroid, so the values computed at
$\alpha = 0$ are identical to their $\alpha$-averages.

\paragraph{Implementation.}
When \texttt{run\_dda.py} detects a spheroid
(\texttt{ab\_ratio == 1} and \texttt{gre\_beta == 0}),
it activates \emph{spheroid mode}:
\begin{itemize}
  \item The DDA is solved only for $N_\beta$ orientations at
        $\alpha = 0$, $\gamma = 0$, with $\cos\beta$ at the
        $N_\beta$ equal-area midpoints (Sec.~\ref{sec:incident-field}).
  \item The full $N_\alpha \times N_\beta \times N_\gamma$ orientation
        grid is then filled analytically using the constraint relations
        \eqref{eq:Ss-constraint}--\eqref{eq:Sp-constraint}:
        \[
          S_s(\alpha_j, \beta_i)
            = A_i + B_i\,e^{i2\alpha_j}, \qquad
          S_p(\alpha_j, \beta_i)
            = A_i - B_i\,e^{i2\alpha_j},
        \]
        where $A_i = [S_s(\beta_i,0) + S_p(\beta_i,0)]/2$ and
        $B_i = [S_s(\beta_i,0) - S_p(\beta_i,0)]/2$.
  \item For backward scattering, the OCBS amplitude follows the same
        rotational structure:
        $S_{\mathrm{bk}}^{\mathrm{OCBS}}(\alpha_j, \beta_i)
         = S_{\mathrm{bk}}^{\mathrm{OCBS}}(0, \beta_i)\,e^{i2\alpha_j}$.
  \item $C_{\mathrm{ext}}$ and $C_{\mathrm{abs}}$ are $\alpha$-invariant,
        so the $\alpha = 0$ values are broadcast to all grid points.
\end{itemize}
This reduces the number of DDA solves from
$N_\alpha \times N_\beta \times N_\gamma$ to $N_\beta$, while
the output grid retains the full
$N_\alpha \times N_\beta \times N_\gamma$ resolution with
\emph{exact} (not interpolated) values at every grid point.

\paragraph{Applicability.}
These relations hold for any particle whose cross-section in the plane
perpendicular to the symmetry axis is circular ($a = b$), regardless
of the axial ratio $b/c$.  They break down for particles lacking a
continuous rotational symmetry axis (e.g., tri-axial ellipsoids with
$a \neq b$, or particles with GRE roughness $\beta_{\mathrm{GRE}} > 0$).

\paragraph{Motivation.}
For spheroids, the $T$-matrix method~\cite{Mishchenko2002} is generally
far more efficient than DDA.  However, the extended-boundary-condition
method (EBCM) used to compute the $T$-matrix becomes numerically
unstable for extreme aspect ratios ($b/c \ll 1$ or $b/c \gg 1$), where
the surface integrals over highly elongated or flattened geometries lead
to severe loss of numerical precision~\cite{Mishchenko2002}.  In such
regimes, DDA with the spheroid-mode optimisation described above provides
a viable alternative: the computation remains stable regardless of the
aspect ratio, and the analytical $\alpha$-averaging reduces the cost by
a factor of $N_\alpha$ relative to the general DDA mode.

\section{Spheroid Parameter Sweep}
\label{sec:spheroid-sweep}
%======================================================================

For systematic studies of spheroid scattering properties, a dedicated
sweep script (\texttt{run\_dda\_spheroid\_sweep.py}) constructs equidistant
grids over three physical parameters and a two-dimensional orientation
grid on a reduced domain, executes the DDA in spheroid mode
(Sec.~\ref{sec:spheroid}), and stores the forward-scattering amplitudes
in an HDF5 file.  The output schema is specified in
\texttt{spheroid\_h5\_schema\_spec.md} and is compatible with the
downstream LUT builder (\texttt{build\_spheroid\_lut.py}).

%----------------------------------------------------------------------
\subsection{Parameter grid}
\label{sec:sweep-grid}
%----------------------------------------------------------------------

The physical-parameter grid is spanned by three equidistant axes:
\begin{align}
  D_{ve}   &\in [D_{ve,\min},\; D_{ve,\max}],
             &  N_{D_{ve}} &\text{ equally spaced points},
  \label{eq:sweep-dve} \\
  n_{\mathrm{real}}
            &\in [n_{\mathrm{real},\min},\;
                   n_{\mathrm{real},\max}],
             &  N_{\mathrm{RI}} &\text{ equally spaced points},
  \label{eq:sweep-ri} \\
  \log_{10}(\mathrm{AR})
            &\in [\log_{10}\mathrm{AR}_{\min},\;
                   \log_{10}\mathrm{AR}_{\max}],
             &  N_{\mathrm{AR}} &\text{ equally spaced points},
  \label{eq:sweep-ar}
\end{align}
where $D_{ve} = 2 r_v$ is the volume-equivalent diameter,
$n_{\mathrm{real}} = \operatorname{Re}(m_p)$ is the real part of the
isotropic particle refractive index, and
$\mathrm{AR} = b/c$ is the aspect ratio of the spheroid
(equal to \texttt{bc\_ratio}; $\mathrm{AR}>1$ oblate, $<1$ prolate).

In addition, a list of $(\lambda_0,\, m_m)$ pairs is specified,
where $\lambda_0$ is the vacuum wavelength and $m_m$ is the medium
refractive index.  Each pair defines an independent scattering problem
stored in its own HDF5 group.  The imaginary refractive index
$\operatorname{Im}(m_p)$ is fixed (default~0).  The total number of
DDA conditions is
$N_\lambda \times N_{D_{ve}} \times N_{\mathrm{RI}} \times N_{\mathrm{AR}}$,
where $N_\lambda$ is the number of wavelength/medium pairs.

The aspect-ratio grid is equally spaced in $\log_{10}(\mathrm{AR})$
so that oblate and prolate shapes are sampled symmetrically about the
sphere ($\mathrm{AR}=1$, i.e.\ $\log_{10}\mathrm{AR}=0$).

%----------------------------------------------------------------------
\subsection{Orientation grid on reduced domain}
\label{sec:sweep-orientation}
%----------------------------------------------------------------------

For spheroids, the body-frame rotation angle $\gamma$ is irrelevant
(Sec.~\ref{sec:spheroid}).  Furthermore, the forward-scattering
amplitudes possess two symmetries that halve the required domain:
\begin{enumerate}
  \item \textbf{Polar mirror symmetry:}
        $\bm{S}(\cos\theta_o) = \bm{S}(-\cos\theta_o)$,
        so only $\cos\theta_o \in [0, 1]$ needs to be stored.
  \item \textbf{Azimuthal periodicity:}
        $\bm{S}(\theta_o,\, \phi_o) = \bm{S}(\theta_o,\, \phi_o + \pi)$,
        so only $\phi_o \in [0, \pi]$ needs to be stored.
\end{enumerate}

The orientation grid is therefore:
\begin{align}
  \cos\theta_{o,j} &= \frac{j}{N_u - 1},
                     \quad j = 0,\dots,N_u-1,
  \label{eq:sweep-costheta} \\
  \phi_{o,i} &= \frac{\pi\,i}{N_\phi - 1},
               \quad i = 0,\dots,N_\phi-1,
  \label{eq:sweep-phi}
\end{align}
where $N_u$ is the number of grid points in $\cos\theta_o \in [0, 1]$
and $N_\phi$ is the number of grid points in $\phi_o \in [0, \pi]$.
Both grids are equidistant and include both endpoints.

The orientation angles map to Euler angles as
$\theta_o \leftrightarrow \beta$ (polar) and
$\phi_o \leftrightarrow \alpha$ (azimuthal).
As described in Sec.~\ref{sec:spheroid}, only $N_u$ DDA solves are
performed (at $\phi_o = 0$); the full $(\cos\theta_o,\, \phi_o)$
grid is populated using the analytical constraint relations.

%----------------------------------------------------------------------
\subsection{HDF5 output structure}
\label{sec:sweep-hdf5}
%----------------------------------------------------------------------

Results are stored in \texttt{dda\_results/dda\_results\_spheroid\_sweep.h5}
using a multi-wavelength group-based layout.  Shared grid axes reside at
the HDF5 root level; per-wavelength data is stored in separate groups.

\begin{itemize}
  \item \textbf{Shared grid axes} (root level, 1D, equidistant):

    \begin{tabular}{@{}llll@{}}
      \toprule
      Dataset & Shape & dtype & Description \\
      \midrule
      \texttt{D\_ve\_grid} & $(N_{D_{ve}},)$ & float64 &
        Volume-equivalent diameter [\si{\micro\metre}] \\
      \texttt{RI\_real\_grid} & $(N_{\mathrm{RI}},)$ & float64 &
        Real refractive index \\
      \texttt{log\_AR\_grid} & $(N_{\mathrm{AR}},)$ & float64 &
        $\log_{10}(\mathrm{AR})$ \\
      \texttt{cos\_theta\_o\_half\_grid} & $(N_u,)$ & float64 &
        $\cos\theta_o$ on $[0,1]$ \\
      \texttt{phi\_o\_grid} & $(N_\phi,)$ & float64 &
        $\phi_o$ on $[0,\pi]$ [rad] \\
      \bottomrule
    \end{tabular}

  \item \textbf{Per-wavelength groups}
        (e.g.\ \texttt{wl\_0p453/}, \texttt{wl\_0p638/},
        \texttt{wl\_0p834/}).
    Group names encode the wavelength with the decimal point replaced by
    \texttt{p}.  Each group carries a required attribute \texttt{wl\_0}
    (float64) storing the exact wavelength.

    Each group contains:
    \begin{itemize}
      \item \textbf{Forward scattering amplitudes}
            (shape $(N_{D_{ve}}, N_{\mathrm{RI}}, N_{\mathrm{AR}},
                     N_u, N_\phi)$, dtype float64):
        \texttt{S\_fw\_theta\_re}, \texttt{S\_fw\_theta\_im},
        \texttt{S\_fw\_phi\_re}, \texttt{S\_fw\_phi\_im}.
        Non-converged cells contain NaN.

      \item \textbf{Convergence flag}
            (shape $(N_{D_{ve}}, N_{\mathrm{RI}}, N_{\mathrm{AR}})$,
            dtype bool):
        \texttt{converged}.
    \end{itemize}

  \item \textbf{Root attributes} (provenance):
    \texttt{m\_m}, \texttt{m\_imag},
    \texttt{rng\_seed}, \texttt{solver\_tol},
    \texttt{block\_dda\_version}.
\end{itemize}

\paragraph{Symmetry handling.}
The HDF5 contains only the reduced-domain data.  The downstream
consumer mirrors $\cos\theta_o$ to $[-1, 1]$ and extends $\phi_o$ to
$[0, 2\pi)$.  The analytical expansion from $\beta$-only DDA solves
to the full $(\cos\theta_o,\, \phi_o)$ grid is performed by
\texttt{block-DDA\_Py} before writing.

An interactive notebook for browsing the file overview and retrieving
results is provided in \texttt{read\_spheroid\_sweep.ipynb}.

\section{Mie Theory Reference}
\label{sec:mie}
%======================================================================

For validation, the DDA solution for a volume-equivalent sphere is
compared to the exact Mie theory~\cite{Bohren1983}.

%----------------------------------------------------------------------
\subsection{Partial-wave expansion}
\label{sec:mie-expansion}
%----------------------------------------------------------------------

For a homogeneous sphere of radius $r_p$, the size parameter is
$x = k\,r_p = 2\pi\,m_m\,r_p / \lambda_0$ and the relative refractive
index is $m_r = m_p / m_m$.
The number of partial-wave terms is
\begin{equation}
  n_{\max} = \lfloor |x + 4\,x^{1/3} + 2| \rfloor.
  \label{eq:nstop}
\end{equation}

The Mie coefficients $a_n$ and $b_n$ are computed from
Riccati--Bessel functions $\psi_n$, $\chi_n$, $\xi_n = \psi_n + i\chi_n$
and the logarithmic derivative $D_n(m_r x)$ via downward
recurrence~\cite{Bohren1983}.

%----------------------------------------------------------------------
\subsection{Efficiencies and scattering amplitudes}
\label{sec:mie-efficiencies}
%----------------------------------------------------------------------

\begin{align}
  Q_{\text{ext}} &= \frac{2}{x^2}\sum_{n=1}^{n_{\max}}
                     (2n+1)\,\operatorname{Re}(a_n + b_n),
  \label{eq:Q-ext-mie} \\
  Q_{\text{sca}} &= \frac{2}{x^2}\sum_{n=1}^{n_{\max}}
                     (2n+1)\,(|a_n|^2 + |b_n|^2),
  \label{eq:Q-sca-mie} \\
  Q_{\text{abs}} &= Q_{\text{ext}} - Q_{\text{sca}}.
  \label{eq:Q-abs-mie}
\end{align}

The angular scattering amplitudes are~\cite{Bohren1983}
\begin{align}
  S_1(\theta)
    &= \sum_{n=1}^{n_{\max}} \frac{2n+1}{n(n+1)}
       \bigl(a_n\,\pi_n(\cos\theta) + b_n\,\tau_n(\cos\theta)\bigr),
  \label{eq:S1-mie} \\
  S_2(\theta)
    &= \sum_{n=1}^{n_{\max}} \frac{2n+1}{n(n+1)}
       \bigl(a_n\,\tau_n(\cos\theta) + b_n\,\pi_n(\cos\theta)\bigr),
  \label{eq:S2-mie}
\end{align}
where $\pi_n$ and $\tau_n$ are the angular functions computed via
upward recurrence.
The forward and backward scattering amplitudes used for validation
are obtained from $S_1$ and $S_2$ at $\theta = 0$ and $\theta = \pi$.
